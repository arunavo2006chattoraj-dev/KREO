import cv2
import numpy as np
import os
import time
import onnxruntime as ort

from insightface.app import FaceAnalysis


# ============================================================
# AUTHENTICATION DATASET
# ============================================================

# CHANGE ONLY THIS NAME
#
# Example:
# USER_NAME = "pp"
# USER_NAME = "john"
# USER_NAME = "alice"
#
# The program will use:
# face_database/<USER_NAME>/embeddings.npy

USER_NAME = "model"


# ============================================================
# PROJECT SETTINGS
# ============================================================

EMBEDDING_FILENAME = "embeddings.npy"

# Authentication threshold
THRESHOLD = 0.35

# Camera settings
CAMERA_INDEX = 0
CAMERA_WIDTH = 640
CAMERA_HEIGHT = 480
CAMERA_FPS = 30

# InsightFace
MODEL_NAME = "buffalo_l"
DETECTION_SIZE = (640, 640)

# InsightFace model storage
MODEL_FOLDER = "insightface_models"


# ============================================================
# PROJECT PATHS
# ============================================================

BASE_DIR = os.path.dirname(
    os.path.abspath(__file__)
)


# Face database
DATABASE_PATH = os.path.join(
    BASE_DIR,
    "face_database"
)


# Selected user's folder
USER_PATH = os.path.join(
    DATABASE_PATH,
    USER_NAME
)


# Selected user's embedding file
EMBEDDING_PATH = os.path.join(
    USER_PATH,
    EMBEDDING_FILENAME
)


# InsightFace model root
MODEL_ROOT = os.path.join(
    BASE_DIR,
    MODEL_FOLDER
)


# ============================================================
# PROJECT INFORMATION
# ============================================================

print("=" * 60)
print("FACE AUTHENTICATION")
print("=" * 60)

print("\nSelected user:")
print(USER_NAME)

print("\nEmbedding file:")
print(EMBEDDING_PATH)

print("\nModel root:")
print(MODEL_ROOT)


# ============================================================
# GPU / ONNX RUNTIME CHECK
# ============================================================

print("\n" + "=" * 60)
print("GPU / ONNX RUNTIME CHECK")
print("=" * 60)


providers = ort.get_available_providers()


print("Available providers:")
print(providers)


if "CUDAExecutionProvider" in providers:

    print("\nCUDAExecutionProvider detected.")
    print("RTX GPU will be used.")

    INSIGHTFACE_PROVIDERS = [
        "CUDAExecutionProvider",
        "CPUExecutionProvider"
    ]

else:

    print("\nWARNING:")
    print("CUDAExecutionProvider is NOT available.")
    print("InsightFace will run on CPU.")

    INSIGHTFACE_PROVIDERS = [
        "CPUExecutionProvider"
    ]


# ============================================================
# CHECK DATABASE FOLDER
# ============================================================

if not os.path.exists(DATABASE_PATH):

    print("\n" + "=" * 60)
    print("ERROR: FACE DATABASE NOT FOUND")
    print("=" * 60)

    print("\nExpected database:")
    print(DATABASE_PATH)

    print("\nExpected structure:")
    print(
        """
face_authentication/
    face_database/
        pp/
            embeddings.npy

        john/
            embeddings.npy
"""
    )

    exit()


# ============================================================
# CHECK SELECTED USER FOLDER
# ============================================================

if not os.path.exists(USER_PATH):

    print("\n" + "=" * 60)
    print("ERROR: USER DATASET NOT FOUND")
    print("=" * 60)

    print("\nSelected user:")
    print(USER_NAME)

    print("\nExpected folder:")
    print(USER_PATH)

    print("\nAvailable datasets:")

    for item in os.listdir(DATABASE_PATH):

        item_path = os.path.join(
            DATABASE_PATH,
            item
        )

        if os.path.isdir(item_path):

            print(
                "  -",
                item
            )

    exit()


# ============================================================
# CHECK EMBEDDING FILE
# ============================================================

if not os.path.exists(EMBEDDING_PATH):

    print("\n" + "=" * 60)
    print("ERROR: EMBEDDING FILE NOT FOUND")
    print("=" * 60)

    print("\nExpected file:")
    print(EMBEDDING_PATH)

    print("\nSelected user:")
    print(USER_NAME)

    print("\nExpected folder structure:")
    print(
        """
face_authentication/
    face_database/
        """
        + USER_NAME
        + """/
            """
        + EMBEDDING_FILENAME
    )

    exit()


# ============================================================
# LOAD REGISTERED EMBEDDINGS
# ============================================================

print("\n" + "=" * 60)
print("LOADING REGISTERED EMBEDDINGS")
print("=" * 60)


try:

    stored_embeddings = np.load(
        EMBEDDING_PATH
    )

except Exception as e:

    print("\nERROR: Could not load embedding file.")

    print("\nDetails:")
    print(e)

    exit()


# ============================================================
# CHECK EMBEDDING SHAPE
# ============================================================

if stored_embeddings.ndim == 1:

    stored_embeddings = (
        stored_embeddings.reshape(1, -1)
    )


print("\nRegistered embeddings loaded.")

print("Embedding shape:")
print(stored_embeddings.shape)

print("Number of registered samples:")
print(len(stored_embeddings))


# ============================================================
# NORMALIZE STORED EMBEDDINGS
# ============================================================

stored_norms = np.linalg.norm(
    stored_embeddings,
    axis=1,
    keepdims=True
)


# Prevent division by zero
stored_norms[
    stored_norms == 0
] = 1.0


stored_embeddings = (
    stored_embeddings /
    stored_norms
)


stored_embeddings = stored_embeddings.astype(
    np.float32
)


# ============================================================
# LOAD INSIGHTFACE
# ============================================================

print("\n" + "=" * 60)
print("LOADING INSIGHTFACE")
print("=" * 60)


try:

    app = FaceAnalysis(
        name=MODEL_NAME,
        root=MODEL_ROOT,
        providers=INSIGHTFACE_PROVIDERS
    )


    app.prepare(
        ctx_id=0,
        det_size=DETECTION_SIZE
    )


except Exception as e:

    print("\nERROR: Could not load InsightFace.")

    print("\nDetails:")
    print(e)

    print("\nExpected model location:")

    print(
        os.path.join(
            MODEL_ROOT,
            "models",
            MODEL_NAME
        )
    )

    exit()


print("\nInsightFace loaded successfully.")


# ============================================================
# OPEN WEBCAM
# ============================================================

print("\n" + "=" * 60)
print("OPENING CAMERA")
print("=" * 60)


cap = cv2.VideoCapture(
    CAMERA_INDEX
)


if not cap.isOpened():

    print("\nERROR: Could not open webcam.")

    exit()


# ============================================================
# CAMERA SETTINGS
# ============================================================

cap.set(
    cv2.CAP_PROP_FRAME_WIDTH,
    CAMERA_WIDTH
)


cap.set(
    cv2.CAP_PROP_FRAME_HEIGHT,
    CAMERA_HEIGHT
)


cap.set(
    cv2.CAP_PROP_FPS,
    CAMERA_FPS
)


actual_width = cap.get(
    cv2.CAP_PROP_FRAME_WIDTH
)


actual_height = cap.get(
    cv2.CAP_PROP_FRAME_HEIGHT
)


actual_camera_fps = cap.get(
    cv2.CAP_PROP_FPS
)


print("Camera width :", actual_width)
print("Camera height:", actual_height)
print("Camera FPS   :", actual_camera_fps)


# ============================================================
# FPS VARIABLES
# ============================================================

previous_time = time.perf_counter()

fps = 0.0

frame_count = 0


# ============================================================
# AUTHENTICATION LOOP
# ============================================================

print("\n" + "=" * 60)
print("CAMERA STARTED")
print("=" * 60)

print("\nAuthenticating against dataset:")
print(USER_NAME)

print("\nLook at the camera.")

print("Press Q to exit.")

print("=" * 60)


while True:

    # ========================================================
    # READ CAMERA FRAME
    # ========================================================

    ret, frame = cap.read()


    if not ret:

        print(
            "ERROR: Could not read camera frame."
        )

        break


    frame_count += 1


    # ========================================================
    # FACE DETECTION + EMBEDDING
    # ========================================================

    ai_start = time.perf_counter()


    faces = app.get(
        frame
    )


    ai_end = time.perf_counter()


    ai_time = (
        ai_end -
        ai_start
    )


    # ========================================================
    # FPS CALCULATION
    # ========================================================

    current_time = time.perf_counter()


    elapsed = (
        current_time -
        previous_time
    )


    if elapsed > 0:

        current_fps = (
            1.0 /
            elapsed
        )


        if fps == 0:

            fps = current_fps

        else:

            fps = (
                0.9 * fps +
                0.1 * current_fps
            )


    previous_time = current_time


    # ========================================================
    # DISPLAY FRAME
    # ========================================================

    display_frame = frame.copy()


    # ========================================================
    # NO FACE
    # ========================================================

    if len(faces) == 0:

        cv2.putText(
            display_frame,
            "NO FACE",
            (30, 40),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.8,
            (0, 0, 255),
            2
        )


    # ========================================================
    # MULTIPLE FACES
    # ========================================================

    elif len(faces) > 1:

        cv2.putText(
            display_frame,
            "ONLY ONE PERSON",
            (30, 40),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.8,
            (0, 0, 255),
            2
        )


        # Draw boxes around all detected faces
        for detected_face in faces:

            x1, y1, x2, y2 = (
                detected_face.bbox.astype(int)
            )


            cv2.rectangle(
                display_frame,
                (x1, y1),
                (x2, y2),
                (0, 0, 255),
                2
            )


    # ========================================================
    # ONE FACE
    # ========================================================

    else:

        face = faces[0]


        # ----------------------------------------------------
        # FACE BOUNDING BOX
        # ----------------------------------------------------

        x1, y1, x2, y2 = (
            face.bbox.astype(int)
        )


        # ----------------------------------------------------
        # LIVE FACE EMBEDDING
        # ----------------------------------------------------

        live_embedding = (
            face.embedding.copy()
        )


        live_norm = np.linalg.norm(
            live_embedding
        )


        # ----------------------------------------------------
        # DEFAULT SIMILARITY
        # ----------------------------------------------------

        similarity = 0.0


        # ----------------------------------------------------
        # NORMALIZE LIVE EMBEDDING
        # ----------------------------------------------------

        if live_norm != 0:

            live_embedding = (
                live_embedding /
                live_norm
            )


            # ------------------------------------------------
            # COSINE SIMILARITY
            # ------------------------------------------------
            #
            # Compare live face against ALL registered
            # samples belonging to the SELECTED USER.
            #
            # Example:
            #
            # USER_NAME = "pp"
            #
            # Only:
            #
            # face_database/pp/embeddings.npy
            #
            # is used here.
            # ------------------------------------------------

            similarities = np.dot(
                stored_embeddings,
                live_embedding
            )


            # Best similarity among the selected
            # user's registered samples

            similarity = float(
                np.max(similarities)
            )


        # ====================================================
        # AUTHENTICATION
        # ====================================================

        if similarity >= THRESHOLD:

            status = (
                f"AUTHORIZED: {USER_NAME}"
            )


            box_color = (
                0,
                255,
                0
            )


        else:

            status = "ACCESS DENIED"


            box_color = (
                0,
                0,
                255
            )


        # ====================================================
        # FACE BOX
        # ====================================================

        cv2.rectangle(
            display_frame,
            (x1, y1),
            (x2, y2),
            box_color,
            2
        )


        # ====================================================
        # STATUS
        # ====================================================

        cv2.putText(
            display_frame,
            status,
            (
                x1,
                max(
                    y1 - 10,
                    30
                )
            ),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.8,
            box_color,
            2
        )


        # ====================================================
        # SIMILARITY
        # ====================================================

        cv2.putText(
            display_frame,
            f"Similarity: {similarity:.3f}",
            (30, 75),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.7,
            box_color,
            2
        )


    # ========================================================
    # PERFORMANCE INFORMATION
    # ========================================================

    cv2.putText(
        display_frame,
        f"FPS: {fps:.1f}",
        (30, 110),
        cv2.FONT_HERSHEY_SIMPLEX,
        0.7,
        (255, 255, 0),
        2
    )


    cv2.putText(
        display_frame,
        f"AI: {ai_time * 1000:.0f} ms",
        (30, 145),
        cv2.FONT_HERSHEY_SIMPLEX,
        0.7,
        (255, 255, 0),
        2
    )


    # ========================================================
    # SELECTED DATASET
    # ========================================================

    cv2.putText(
        display_frame,
        f"Dataset: {USER_NAME}",
        (30, 180),
        cv2.FONT_HERSHEY_SIMPLEX,
        0.7,
        (255, 255, 0),
        2
    )


    # ========================================================
    # SHOW CAMERA
    # ========================================================

    cv2.imshow(
        "GPU Face Authentication",
        display_frame
    )


    # ========================================================
    # QUIT
    # ========================================================

    key = cv2.waitKey(1) & 0xFF


    if key == ord("q"):

        break


# ============================================================
# CLEANUP
# ============================================================

cap.release()

cv2.destroyAllWindows()


print("\nAuthentication stopped.")
